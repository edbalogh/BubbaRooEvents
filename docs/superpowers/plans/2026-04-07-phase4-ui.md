```jsx
import { useState } from 'react';

/**
 * Shows a "?" icon next to a field value when there are conflicting values
 * from different sources. Hover shows all conflicting values.
 *
 * Props:
 *   conflicts: array of {source: string, value: any} — the conflicting values
 *   fieldLabel: string — human-readable field name for the tooltip ("Price", "Date", etc.)
 */
export function ConflictIndicator({ conflicts, fieldLabel }) {
  const [visible, setVisible] = useState(false);

  if (!conflicts || conflicts.length < 2) return null;

  return (
    <span
      style={{ position: 'relative', display: 'inline-block', marginLeft: 4 }}
      onMouseEnter={() => setVisible(true)}
      onMouseLeave={() => setVisible(false)}
    >
      <span
        style={{
          cursor: 'help',
          color: '#888',
          fontSize: '0.75rem',
          border: '1px solid #ccc',
          borderRadius: '50%',
          padding: '0 4px',
          userSelect: 'none',
        }}
      >
        ?
      </span>
      {visible && (
        <div
          style={{
            position: 'absolute',
            bottom: '100%',
            left: '50%',
            transform: 'translateX(-50%)',
            background: '#333',
            color: '#fff',
            borderRadius: 4,
            padding: '6px 10px',
            fontSize: '0.75rem',
            whiteSpace: 'nowrap',
            zIndex: 100,
            marginBottom: 4,
          }}
        >
          <strong>{fieldLabel} across sources:</strong>
          {conflicts.map((c, i) => (
            <div key={i}>
              {c.source}: {String(c.value)}
            </div>
          ))}
        </div>
      )}
    </span>
  );
}
```

### Task 5: Create SourceRow component

**Files:**
- Create: `frontend/src/components/SourceRow.jsx`

- [ ] **Step 1: Create the component**

Create `frontend/src/components/SourceRow.jsx`:

```jsx
import { useState } from 'react';
import { refreshSource, getRefreshStatus, updateSourcePreference } from '../api/sources';

/**
 * A single row in the Sources table.
 * Props:
 *   source: {slug, name, source_type, coverage_cities, last_scraped_at, scrape_status, discovery_confidence}
 *   onRefreshed: callback() — called when refresh completes
 */
export function SourceRow({ source, onRefreshed }) {
  const [refreshing, setRefreshing] = useState(false);
  const [paused, setPaused] = useState(source.scrape_status === 'paused');

  const handleRefresh = async () => {
    setRefreshing(true);
    try {
      const { job_id } = await refreshSource(source.slug);
      // Poll until complete
      const poll = setInterval(async () => {
        const status = await getRefreshStatus(source.slug, job_id);
        if (status.status === 'SUCCESS' || status.status === 'FAILURE') {
          clearInterval(poll);
          setRefreshing(false);
          onRefreshed?.();
        }
      }, 2000);
    } catch {
      setRefreshing(false);
    }
  };

  const handlePauseToggle = async () => {
    const newPref = paused ? 'active' : 'disabled';
    await updateSourcePreference(source.slug, newPref);
    setPaused(!paused);
  };

  const lastScraped = source.last_scraped_at
    ? new Date(source.last_scraped_at).toLocaleString()
    : 'Never';

  const statusColor = { active: '#4caf50', paused: '#888', error: '#f44336' }[source.scrape_status] || '#888';

  return (
    <tr>
      <td>{source.name}</td>
      <td>{source.coverage_cities || 'All cities'}</td>
      <td>{source.source_type}</td>
      <td>{lastScraped}</td>
      <td>
        <span style={{ color: statusColor }}>⬤</span>{' '}
        {paused ? 'paused' : source.scrape_status}
      </td>
      <td>
        <button onClick={handleRefresh} disabled={refreshing} style={{ marginRight: 8 }}>
          {refreshing ? '⏳ Refreshing...' : '↺ Refresh'}
        </button>
        <button onClick={handlePauseToggle}>
          {paused ? '▶ Resume' : '⏸ Pause'}
        </button>
      </td>
    </tr>
  );
}
```

```jsx
import { useState, useEffect } from 'react';
import { SourceRow } from '../components/SourceRow';
import {
  fetchSources,
  runDiscovery,
  getDiscoveryStatus,
  suggestSource,
} from '../api/sources';

export function SourcesPage() {
  const [sources, setSources] = useState([]);
  const [discoveryStatus, setDiscoveryStatus] = useState(null);
  const [discovering, setDiscovering] = useState(false);
  const [suggestUrl, setSuggestUrl] = useState('');
  const [suggestOpen, setSuggestOpen] = useState(false);
  const [loading, setLoading] = useState(true);

  const loadData = async () => {
    const [srcs, status] = await Promise.all([fetchSources(), getDiscoveryStatus()]);
    setSources(srcs);
    setDiscoveryStatus(status);
    setLoading(false);
  };

  useEffect(() => { loadData(); }, []);

  const handleRunDiscovery = async () => {
    setDiscovering(true);
    await runDiscovery();
    // Poll discovery status — simplified: just reload after 10s
    setTimeout(async () => {
      await loadData();
      setDiscovering(false);
    }, 10000);
  };

  const handleSuggest = async () => {
    if (!suggestUrl.trim()) return;
    await suggestSource(suggestUrl.trim());
    setSuggestUrl('');
    setSuggestOpen(false);
    await loadData();
  };

  if (loading) return <div>Loading sources...</div>;

  return (
    <div style={{ padding: '24px' }}>
      <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: 16 }}>
        <h1>Event Sources</h1>
        <div>
          <span style={{ marginRight: 16, color: '#888', fontSize: '0.85rem' }}>
            Last discovery:{' '}
            {discoveryStatus?.last_discovery_at
              ? new Date(discoveryStatus.last_discovery_at).toLocaleString()
              : 'Never'}
            {' '}({discoveryStatus?.total_discovered_sources || 0} discovered)
          </span>
          <button onClick={handleRunDiscovery} disabled={discovering} style={{ marginRight: 8 }}>
            {discovering ? '🔍 Discovering...' : '🔍 Run Discovery Now'}
          </button>
          <button onClick={() => setSuggestOpen(true)}>+ Suggest Source</button>
        </div>
      </div>

      {suggestOpen && (
        <div style={{ marginBottom: 16, padding: 12, background: '#f5f5f5', borderRadius: 4 }}>
          <input
            type="url"
            placeholder="https://yourcityevents.com/calendar"
            value={suggestUrl}
            onChange={e => setSuggestUrl(e.target.value)}
            style={{ width: 400, marginRight: 8, padding: '4px 8px' }}
          />
          <button onClick={handleSuggest} style={{ marginRight: 8 }}>Submit</button>
          <button onClick={() => setSuggestOpen(false)}>Cancel</button>
        </div>
      )}

      <table style={{ width: '100%', borderCollapse: 'collapse' }}>
        <thead>
          <tr style={{ textAlign: 'left', borderBottom: '2px solid #eee' }}>
            <th style={{ padding: '8px' }}>Name</th>
            <th style={{ padding: '8px' }}>City</th>
            <th style={{ padding: '8px' }}>Type</th>
            <th style={{ padding: '8px' }}>Last Scraped</th>
            <th style={{ padding: '8px' }}>Status</th>
            <th style={{ padding: '8px' }}>Actions</th>
          </tr>
        </thead>
        <tbody>
          {sources.map(source => (
            <SourceRow key={source.slug} source={source} onRefreshed={loadData} />
          ))}
        </tbody>
      </table>

      {sources.length === 0 && (
        <div style={{ textAlign: 'center', color: '#888', padding: 32 }}>
          No sources yet. Run discovery to find local event sites.
        </div>
      )}
    </div>
  );
}
```

```jsx
import { useState, useEffect } from 'react';
import { fetchFlaggedDuplicates, mergeDuplicates } from '../api/duplicates';

export function DuplicatesPage() {
  const [duplicates, setDuplicates] = useState([]);
  const [loading, setLoading] = useState(true);

    const data = await fetchFlaggedDuplicates();
    setDuplicates(data);

          key={dup.id}
          style={{ border: '1px solid #eee', borderRadius: 4, padding: 16, marginBottom: 16 }}
```
